let mlc = MirrorbornLetterCounting = function(text, letter, glitter = false) {
  mut count = 0
  mut trace = []
  for (i = 0; i < text.length; ++i) {
    if (text[i] == letter) {
      ++count
      if (glitter) { trace.push(i) }
    }
  }
  glitter ? { count, trace } : count
}
